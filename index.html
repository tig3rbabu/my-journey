    <script>

        const KEYS = ['assets', 'serotonin', 'health', 'nic', 'mj', 'alc', 'food', 'bull'];

        let myChart = null;


        function getFaderColor(p) {

            let r, g;

            if (p < 50) { r = 255; g = Math.round((p / 50) * 165); } 

            else { r = Math.round(255 - ((p - 50) / 50) * 255); g = Math.round(165 + ((p - 50) / 50) * 90); }

            return `rgb(${r},${g},0)`;

        }


        function updateUI() {

            KEYS.forEach(key => {

                const val = document.getElementById(`in-${key}`).value;

                const color = getFaderColor(val);

                const group = document.getElementById(`n-${key}`);

                

                // Color the bloom

                group.querySelector('.flare-bloom').style.fill = color;

                

                // Morph icons

                const light = group.querySelector('.icon-light');

                const shadow = group.querySelector('.icon-shadow');

                

                // Opacity logic

                const lightOpacity = val / 100;

                const shadowOpacity = 1 - (val / 100);

                

                light.style.opacity = lightOpacity;

                shadow.style.opacity = shadowOpacity;

                

                light.style.stroke = color;

                shadow.style.stroke = color;

                

                // Shadow icons get a slight flicker/blur if they are visible

                shadow.style.filter = shadowOpacity > 0.5 ? `drop-shadow(0 0 5px ${color})` : 'none';

            });

            const state = {};

            KEYS.forEach(k => state[k] = document.getElementById(`in-${k}`).value);

            localStorage.setItem('morph_balance_v3', JSON.stringify(state));

        }


        function logSnapshot() {

            const history = JSON.parse(localStorage.getItem('morph_history_v3') || '[]');

            const snap = { t: new Date().toLocaleTimeString([], {hour: '2-digit', minute:'2-digit'}) };

            KEYS.forEach(k => snap[k] = document.getElementById(`in-${k}`).value);

            history.push(snap);

            if(history.length > 40) history.shift();

            localStorage.setItem('morph_history_v3', JSON.stringify(history));

        }


        function renderChart() {

            const history = JSON.parse(localStorage.getItem('morph_history_v3') || '[]');

            const ctx = document.getElementById('mainChart').getContext('2d');

            if(myChart) myChart.destroy();

            myChart = new Chart(ctx, {

                type: 'line',

                data: {

                    labels: history.map(h => h.t),

                    datasets: KEYS.map(k => ({

                        data: history.map(h => h[k]),

                        borderColor: getFaderColor(history[history.length-1]?.[k] || 50),

                        borderWidth: 2, pointRadius: 0, tension: 0.4

                    }))

                },

                options: { responsive: true, plugins: { legend: { display: false } }, scales: { y: { display: false }, x: { display: false } } }

            });

        }


        function showPage(p) {

            document.getElementById('page-home').classList.toggle('hidden', p !== 'home');

            document.getElementById('page-stats').classList.toggle('hidden', p !== 'stats');

            document.getElementById('nav-home').classList.toggle('tab-active', p === 'home');

            document.getElementById('nav-stats').classList.toggle('tab-active', p === 'stats');

            if(p === 'stats') renderChart();

        }


        lucide.createIcons();

        const saved = JSON.parse(localStorage.getItem('morph_balance_v3') || '{"assets":50,"serotonin":50,"health":50,"nic":50,"mj":50,"alc":50,"food":50,"bull":50}');

        KEYS.forEach(k => document.getElementById(`in-${k}`).value = saved[k]);

        setTimeout(updateUI, 100);
